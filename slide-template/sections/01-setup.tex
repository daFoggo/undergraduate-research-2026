\section{Setup}

\begin{frame}{Recap: previous meeting}
  \begin{itemize}
    \item \alert{Experiments:} Meeting Room (TV) and, newly added, Kitchen
          (Fridge, Microwave).
    \item \alert{Fact $\rightarrow$ Case:} a collection of sensor observations,
          contextual information, and semantic annotations relevant to the
          inquiry goal.
    \item \alert{Signature extraction $\rightarrow$ Cycle detection:} an
          algorithmic rule that scans sensor traces and returns the temporal
          boundaries $[t_{\mathrm{start}}, t_{\mathrm{end}}]$ of an active
          service episode or appliance cycle.
  \end{itemize}
\end{frame}

\begin{frame}{Inquiry goals and indicators}
  \footnotesize
  \setlength{\tabcolsep}{3pt}

    \textbf{Inquiry} & \textbf{Goal $\Gamma$} & \textbf{Indicators $I$} \\
    \hline
    Fridge & Confirm the detector draws consistent on/off boundaries across many
      repetitions (not anomaly detection).
      & energy, duration, mean power, peak power \\
    Microwave & Handle a sparse, short, noisy signal, and sort each burst into one
      of four uses.
      & energy, duration, peak power \\
    Session & Measure real meeting-room occupancy rather than inferring it from the
      TV alone.
      & energy, duration, mean power \\
    Comfort & Trade comfort against energy, and surface AC left running when nobody
      is in the room.
      & avg. temperature, avg. humidity \\
